cmake_minimum_required(VERSION 3.8)
project(pedsim_original)

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

option(SHALL_DEBUG "Enable debug features" OFF)
option(SHALL_PROFILE "Enable the code profiling feature" OFF)

add_compile_options(-Wall -Wunused -pipe)
if(SHALL_DEBUG)
  add_compile_options(-O0 -ggdb -g3)
  add_compile_definitions(DEBUG)
else()
  add_compile_options(-Os)
endif()

find_package(ament_cmake REQUIRED)
find_package(Boost REQUIRED)

set(SOURCES
  src/ped_agent.cpp
  src/ped_angle.cpp
  src/ped_obstacle.cpp
  src/ped_scene.cpp
  src/ped_tree.cpp
  src/ped_vector.cpp
  src/ped_waypoint.cpp
)

add_library(pedsim_original SHARED ${SOURCES})
target_include_directories(pedsim_original PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
  "$<INSTALL_INTERFACE:include>"
)
target_link_libraries(pedsim_original ${Boost_LIBRARIES})

install(
  TARGETS pedsim_original
  EXPORT export_pedsim_original
  LIBRARY DESTINATION lib
  ARCHIVE DESTINATION lib
  RUNTIME DESTINATION bin
)

install(DIRECTORY include/${PROJECT_NAME}/
  DESTINATION include/${PROJECT_NAME}
  PATTERN ".svn" EXCLUDE
)

ament_export_targets(export_pedsim_original HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(Boost)
ament_package()
