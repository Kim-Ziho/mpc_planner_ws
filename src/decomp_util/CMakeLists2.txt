cmake_minimum_required(VERSION 3.8)
project(decomp_util)

if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()

add_compile_options(-Wall -Wno-deprecated-declarations -O3)

set(DECOMP_OLD TRUE)

find_package(ament_cmake REQUIRED)
find_package(eigen3_cmake_module REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(ros_tools REQUIRED)

set(LIBRARY_SOURCES
  src/decomp_basis/data_utils.cpp
  src/decomp_geometry/ellipsoid.cpp
  src/decomp_geometry/geometric_utils.cpp
  src/decomp_geometry/polyhedron.cpp
  src/decomp_base.cpp
  src/ellipsoid_decomp.cpp
  src/iterative_decomp.cpp
  src/line_segment.cpp
  src/seed_decomp.cpp
)

add_library(${PROJECT_NAME} SHARED ${LIBRARY_SOURCES})
target_include_directories(${PROJECT_NAME} PUBLIC
  "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
  "$<INSTALL_INTERFACE:include>"
)
if(DECOMP_OLD)
  target_compile_definitions(${PROJECT_NAME} PUBLIC DECOMP_OLD)
endif()
target_link_libraries(${PROJECT_NAME} Eigen3::Eigen)
ament_target_dependencies(${PROJECT_NAME} ros_tools)

install(
  TARGETS ${PROJECT_NAME}
  EXPORT export_${PROJECT_NAME}
  LIBRARY DESTINATION lib
  ARCHIVE DESTINATION lib
  RUNTIME DESTINATION bin
)

install(DIRECTORY include/${PROJECT_NAME}/
  DESTINATION include/${PROJECT_NAME}
)

ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_include_directories(include)
ament_export_dependencies(ros_tools eigen3_cmake_module Eigen3)
ament_package()
